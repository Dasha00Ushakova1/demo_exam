Модуль 1: База данных и Ключи

Задача: Создать таблицы, настроить связи и импортировать данные.

Шаг 1: Конвертация Excel в CSV

DB Browser не понимает .xlsx. Открой файлы в Excel и сохрани как CSV UTF-8 (разделитель запятая).

Шаг 2: Импорт таблиц

  1. Открой DB Browser, создай базу database.db.
  2. Файл -> Импорт -> Таблица из CSV. Выбери файл.
  3. КРИТИЧНО: Внизу окна смени типы данных. Цена (Price) = REAL, Количество (Quantity) = INTEGER.
  4. Нажми ОК и сохрани изменения.

Шаг 3: Настройка Первичных и Внешних ключей (Связи)

Зачем это нужно? Ключи нужны, чтобы база не удалила категорию, если на неё ссылается товар, и чтобы в заказ нельзя было добавить несуществующий размер обуви.
  1. Перейди во вкладку Database Structure (Структура БД).
  2. Нажми ПКМ по таблице (например, products) -> Изменить таблицу.
  3. Первичный ключ (PK): Дважды кликни по колонке id и поставь галочку PK (Primary Key).
  4. Внешний ключ (FK): Перейди на вкладку Foreign Keys внизу. Нажми Add.
    В колонке Table выбери таблицу-справочник (например, categories).
    В колонке To выбери id.
    В колонке From выбери category_id.
  5. Нажми ОК и Записать изменения.

Шаг 4: Составной ключ (Для склада и заказов)

В таблице stock_items товар может иметь несколько размеров. Чтобы строка была уникальной, ключ делается из двух столбцов.

  1. Открой таблицу на изменение.
  2. Выдели сразу два столбца (например, product_id и size_id), зажав Ctrl.
  3. Внизу нажми кнопку с иконкой ключа Set Primary Key.

Шаг 5: ER-диаграмма (draw.io)

1. В DB Browser: Файл -> Экспорт -> База данных в SQL. Сними галочки с INSERT.

2. Открой сайт draw.io. Меню: Упорядочить -> Вставить -> Расширенные -> SQL. Вставь код. Таблицы появятся сами!

3. Соедини их стрелками "вороньи лапки" (Один-ко-многим).

Шаг 2. Логика каталога (MainWindow.xaml.cs)

public partial class MainWindow : Window {
    private Context db = new Context();
    private string currentRole;
    private string currentUser;
    private bool isAscending = true;

    // Получаем роль из окна логина
    public MainWindow(string role, string fullName) {
        InitializeComponent();
        currentRole = role;
        currentUser = fullName;
        UserNameText.Text = fullName; // В правом верхнем углу
        
        // Роли
        if (role == "Менеджер" || role == "Администратор") {
            // Показываем поиск и сортировку
        } else {
            SearchBox.Visibility = Visibility.Collapsed;
            SupplierBox.Visibility = Visibility.Collapsed;
        }

        LoadData();
    }

    private void LoadData() {
        var query = db.Products.AsQueryable();

        // ФИЛЬТРАЦИЯ (Поиск и поставщик)
        if (!string.IsNullOrEmpty(SearchBox.Text)) {
            query = query.Where(p => p.Name.Contains(SearchBox.Text) || p.Description.Contains(SearchBox.Text));
        }
        if (SupplierBox.SelectedItem != null) {
            query = query.Where(p => p.SupplierId == (int)SupplierBox.SelectedValue);
        }

        // СОРТИРОВКА
        if (isAscending) query = query.OrderBy(p => p.Quantity);
        else query = query.OrderByDescending(p => p.Quantity);

        // ВЫВОД
        var products = query.Select(p => new {
            p.Id,
            ProdName = p.Name,
            p.Description,
            ManufacturerName = p.Manufacturer.Name,
            Price = p.Price,
            FinalPrice = p.Price - (p.Price * p.Discount / 100), // Скидка
            IsDiscounted = p.Discount > 0 ? Visibility.Visible : Visibility.Collapsed,
            p.Quantity,
            ImgPath = string.IsNullOrEmpty(p.ImagePath) ? "/Images/picture.png" : "/Images/Products/" + p.ImagePath,
            
            // Цвета по заданию
            BgColor = p.Quantity == 0 ? "LightBlue" : (p.Discount > 15 ? "#2E8B57" : "White")
        }).ToList();

        ProductList.ItemsSource = products;
    }

    private void Filter_Changed(object sender, EventArgs e) { LoadData(); }
    
    private void SortBtn_Click(object sender, RoutedEventArgs e) {
        isAscending = !isAscending;
        LoadData();
    }
}

private string newImagePath = null;

private void LoadPhoto_Click(object sender, RoutedEventArgs e) {
    OpenFileDialog ofd = new OpenFileDialog();
    ofd.Filter = "Images|*.jpg;*.png";
    if (ofd.ShowDialog() == true) {
        BitmapImage img = new BitmapImage(new Uri(ofd.FileName));
        // ПРОВЕРКА РАЗМЕРА 300х200
        if (img.PixelWidth > 300 || img.PixelHeight > 200) {
            MessageBox.Show("Картинка должна быть 300х200!");
            return;
        }
        newImagePath = ofd.FileName;
        PhotoPreview.Source = img;
    }
}

private void SaveBtn_Click(object sender, RoutedEventArgs e) {
    using (var db = new Context()) {
        var product = db.Products.Find(productId); // Для редактирования
        
        if (newImagePath != null) {
            // Удаляем старую картинку
            if (!string.IsNullOrEmpty(product.ImagePath)) {
                File.Delete(Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "Images/Products/", product.ImagePath));
            }
            // Копируем новую
            string fileName = Guid.NewGuid().ToString() + ".jpg";
            string destPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "Images/Products/", fileName);
            File.Copy(newImagePath, destPath);
            product.ImagePath = fileName;
        }
        
        // Сохраняем данные
        db.SaveChanges();
        this.Close();
    }
}
